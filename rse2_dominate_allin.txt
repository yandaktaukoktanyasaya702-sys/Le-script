local function log(msg)
	table.insert(logLines, 1, os.date("%H:%M:%S") .. "  " .. tostring(msg))
	while #logLines > 8 do table.remove(logLines) end
end

local function invoke(remote, ...)
	if not remote then return nil end
	local ok, res = pcall(remote.InvokeServer, remote, ...)
	if ok and type(res) == "table" then return res end
	return nil
end

local function refresh()
	local b = invoke(Boot)
	if b and b.profile then
		profile = b.profile
		seed = profile.seed
	end
	return profile
end

local function maxLev(level)
	level = tonumber(level) or 1
	if level >= 120 then return 100 end
	if level >= 60 then return 75 end
	if level >= 50 then return 50 end
	if level >= 20 then return 20 end
	if level >= 10 then return 10 end
	if level >= 5 then return 5 end
	return 1
end

local function eligible(a)
	if not a or not a.id then return false end
	if not CLASSES[a.class] then return false end
	if a.meme or a.isMemeCoin or a.isPlayerStock or a.composition or a.track or a.ratio then return false end
	if not State.assetIsProcedural(a) then return false end
	if State.assetIsDelisted(a) then return false end
	if (tonumber(a.vol) or 0) > 2.05 then return false end
	return true
end

local function analyse(a, fromT, horizon, steps)
	local p0 = PE.priceAt(seed, a, fromT)
	if not p0 or p0 <= 0 then return nil end
	local peak, peakT, low = p0, fromT, p0
	local minBeforePeak, lowNow = p0, p0
	for i = 1, steps do
		local t = fromT + horizon * i / steps
		local p = PE.priceAt(seed, a, t)
		if p then
			if p < lowNow then lowNow = p end
			if p > peak then
				peak, peakT = p, t
				minBeforePeak = lowNow
			end
			if p < low then low = p end
		end
	end
	return {
		entry = p0,
		peak = peak,
		peakT = peakT,
		gain = (peak - p0) / p0,
		minBeforePeak = minBeforePeak,
		ddBeforePeak = (p0 - minBeforePeak) / p0,
		minAhead = low,
	}
end

local function equityOf(p)
	local now = State.now()
	local eq = tonumber(p.cash) or 0
	for _, v in ipairs(p.positions or {}) do
		local a = State.resolveAsset(v.id)
		local cur = a and PE.priceAt(seed, a, now) or v.avg
		eq = eq + (tonumber(v.initialMargin) or 0) + (cur - v.avg) * v.qty
	end
	return eq
end

local function impactAt(a, qty, price)
	local ok, v = pcall(PE.executionImpactFor, a, qty, price, 1)
	if ok and type(v) == "number" then return v end
	return 0
end

local function capacity(a, price)
	local floatUnits = (a.shares or a.supply or 0) * 1e6
	return floatUnits * price * 0.0003
end

local function scan()
	local now = State.now()
	local equity = equityOf(profile)
	local levMax = maxLev(profile.level)
	local cash = tonumber(profile.cash) or 0
	local coarse = {}
	for _, a in ipairs(State.listedAssets()) do
		if eligible(a) then
			local p0 = PE.priceAt(seed, a, now)
			if p0 and p0 > 0 then
				local hi = p0
				for i = 1, CONFIG.COARSE do
					local p = PE.priceAt(seed, a, now + CONFIG.HORIZON * i / CONFIG.COARSE)
					if p and p > hi then hi = p end
				end
				local g = (hi - p0) / p0
				if g > CONFIG.MIN_EDGE then table.insert(coarse, { a = a, gain = g }) end
			end
		end
	end
	table.sort(coarse, function(x, y) return x.gain > y.gain end)

	local best
	for i = 1, math.min(8, #coarse) do
		local c = coarse[i]
		local r = analyse(c.a, now, CONFIG.HORIZON, CONFIG.FINE)
		if r and r.gain > CONFIG.MIN_EDGE then
			local levByPath = math.floor(0.9 / (CONFIG.LIQ_SAFETY * math.max(r.ddBeforePeak, 0.008)))
			local lev = math.max(1, math.min(levMax, CONFIG.LEV_CAP, levByPath))
			local cap = capacity(c.a, r.entry)
			local margin = math.min(cash * CONFIG.MARGIN_RISK, equity * CONFIG.MARGIN_RISK)
			if cap > 0 then margin = math.min(margin, cap * 0.95 / lev) end
			local notional = margin * lev
			if notional >= 50 then
				local qty = notional / r.entry
				local imp = impactAt(c.a, qty, r.entry)
				local net = r.gain - 2 * imp
				local holdReal = (r.peakT - now) / CFG.TIME_SCALE
				if net > CONFIG.MIN_NET and holdReal >= 10 then
					local score = net * lev
					if not best or score > best.score then
						best = { a = c.a, plan = r, lev = lev, qty = qty, imp = imp, net = net, score = score, margin = margin }
					end
				end
			end
		end
	end
	return best
end

local function openPosition(pick)
	if not alive() then return false end
	if #(profile.positions or {}) > 0 then return false end
	local plan, a = pick.plan, pick.a
	local qty = pick.qty
	if qty <= 0 then return false end
	if (tonumber(profile.cash) or 0) < (pick.margin or 0) * 1.02 then
		log("skip: cash short")
		return false
	end
	local res = invoke(Trade, { action = "open", id = a.id, qty = qty, direction = 1, lev = pick.lev })
	if not res then log("open: no reply"); return false end
	if not res.ok then
		local e = tostring(res.err)
		log("reject: " .. e)
		local low = e:lower()
		if low:find("slow down") then
			backoff = os.clock() + 60
		else
			cooldown = os.clock() + 8
		end
		return false
	end
	profile = res
	local fresh
	for _, v in ipairs(res.positions or {}) do
		if v.id == a.id and (not fresh or (tonumber(v.openedAt) or 0) >= (tonumber(fresh.openedAt) or 0)) then
			fresh = v
		end
	end
	local peakT = plan.peakT
	local lead = math.max(1800, (peakT - State.now()) * CONFIG.LEAD)
	holding = {
		id = a.id,
		entry = fresh and fresh.avg or plan.entry,
		qty = fresh and fresh.qty or qty,
		lev = fresh and tonumber(fresh.lev) or pick.lev,
		exitT = peakT - lead,
		peakT = peakT,
		peak = plan.peak,
		net = pick.net,
	}
	log(string.format("OPEN %s %dx net+%.1f%% hold %ds",
		a.id, holding.lev, pick.net * 100, math.floor((peakT - lead - State.now()) / CFG.TIME_SCALE)))
	return true
end

local function closePosition(reason)
	if not alive() then return false end
	local res = invoke(Trade, { action = "closeAll" })
	if res and res.ok then
		profile = res
		log("CLOSE " .. tostring(reason) .. " +$" .. string.format("%.0f", tonumber(res.sellProfit) or 0))
		holding = nil
		cooldown = os.clock() + CONFIG.COOLDOWN
		return true
	end
	local e = tostring(res and res.err)
	local low = e:lower()
	if low:find("slow down") then
		backoff = os.clock() + 60
		log("close fail: slow down")
	elseif low:find("no open positions") then
		holding = nil
		refresh()
		log("closed externally")
	else
		log("close fail: " .. e)
	end
	return false
end

local function redeemAll()
	for _, code in ipairs(CODES) do invoke(Redeem, code) end
	refresh()
end

local function claimAll()
	local function t(r, ...) invoke(r, ...) end
	for _, k in ipairs({ "group", "favorite", "like" }) do t(ClaimReward, k) end
	t(ClaimDaily); t(ClaimDesk); t(ClaimCareer); t(ClaimCloseReport); t(FreeSkip)
	refresh()
end

local function fmt(n)
	n = tonumber(n) or 0
	local neg = n < 0
	n = math.abs(n)
	local s
	if n >= 1e9 then s = string.format("%.2fB", n / 1e9)
	elseif n >= 1e6 then s = string.format("%.2fM", n / 1e6)
	elseif n >= 1e3 then s = string.format("%.1fK", n / 1e3)
	else s = string.format("%.0f", n) end
	return (neg and "-" or "") .. s
end

task.spawn(function()
	pcall(function()
		refresh()
		sessionEquity = equityOf(profile)
		log("online seed " .. tostring(seed))
	end)
	task.wait(0.5)
	task.spawn(function() pcall(redeemAll) end)
	task.wait(0.5)
	task.spawn(function() pcall(claimAll) end)
	while alive() do
		task.wait(300)
		if running and alive() then
			task.spawn(function() pcall(function() claimAll(); redeemAll() end) end)
		end
	end
end)

task.spawn(function()
	while alive() do
		task.wait(CONFIG.POLL)
		if not alive() then break end
		local ok, err = pcall(function()
			if not profile or os.clock() - lastRefresh > 10 then
				refresh()
				lastRefresh = os.clock()
			end
			if not profile or not seed then status.Text = "no profile"; return end

			local positions = profile.positions or {}
			if holding and #positions == 0 then
				holding = nil
				cooldown = os.clock() + 5
			end
			if not holding and #positions > 0 then
				local v = positions[1]
				local a = State.resolveAsset(v.id)
				if a then
					local now2 = State.now()
					local plan = analyse(a, now2, CONFIG.HORIZON, CONFIG.FINE)
					holding = {
						id = v.id, entry = v.avg, qty = v.qty, lev = v.lev,
						exitT = (plan and plan.peakT - math.max(1800, (plan.peakT - now2) * CONFIG.LEAD)) or (now2 + 3600),
						peakT = plan and plan.peakT or (now2 + 3600),
						peak = plan and plan.peak, net = plan and plan.gain or 0,
					}
					log("adopted " .. v.id)
				end
			end

			local now = State.now()
			if holding then
				local a = State.resolveAsset(holding.id)
				local cur = a and PE.priceAt(seed, a, now) or holding.entry
				local pnl = (cur - holding.entry) / holding.entry * holding.lev * 100
				local liq = holding.entry * (1 - 0.9 / holding.lev)
				local safety = (cur - liq) / cur

				local exitNow, why = false, "peak"
				if cur <= liq * 1.03 then
					exitNow, why = true, "safety"
				elseif holding.peak and cur >= holding.peak * 0.995 then
					exitNow, why = true, "peak"
				elseif holding.peakT and now >= holding.peakT - 0.5 * CFG.TIME_SCALE then
					exitNow, why = true, "peak"
				end

				if exitNow then
					closePosition(why)
				else
					local toPeak = holding.peakT and math.max(0, (holding.peakT - now) / CFG.TIME_SCALE) or 0
					local left = holding.peak and math.max(0, (holding.peak - cur) / cur * 100) or 0
					status.Text = string.format("%s %dx %+.1f%% peak +%.1f%% in %ds liq %+.0f%%",
						holding.id, holding.lev, pnl, left, math.floor(toPeak), safety * 100)
				end
			elseif running and os.clock() >= cooldown and os.clock() >= backoff then
				status.Text = "scanning"
				local pick = scan()
				if pick then openPosition(pick) else status.Text = "no edge" end
			else
				status.Text = "waiting"
			end

			local eq = equityOf(profile)
			local delta = eq - (sessionEquity or eq)
			stats.Text = string.format("equity $%s   %s$%s\nlvl %d  xp %d  cash $%s",
				fmt(eq), delta >= 0 and "+" or "-", fmt(math.abs(delta)),
				tonumber(profile.level) or 1, tonumber(profile.xp) or 0, fmt(profile.cash))
			logLabel.Text = table.concat(logLines, "\n")
		end)
		if not ok then log("ERR " .. tostring(err)) end
	end
end)
