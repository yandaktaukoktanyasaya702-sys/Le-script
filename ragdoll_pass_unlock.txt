end

end

end

local function refresh()
	local count = 0
	for _, passName in ipairs(ORDER) do
		local on = owned(passIds[passName])
		if on then
			count += 1
		end
		local row = rows[passName]
		if row then
			row.state.Text = on and "SPOOFED" or "off"
			row.state.TextColor3 = on and Color3.fromRGB(120, 220, 140) or Color3.fromRGB(120, 126, 136)
			row.button.BackgroundColor3 = on and Color3.fromRGB(34, 50, 40) or Color3.fromRGB(32, 36, 42)
		end
	end
	footer.Text = ("spoofed %d/%d · local only, server untouched"):format(count, #ORDER)
end

local function updateHold()
	if hold then
		holdBtn.Text = "hold: ON — re-asserts if the game overwrites"
		holdBtn.BackgroundColor3 = Color3.fromRGB(32, 46, 38)
		holdBtn.TextColor3 = Color3.fromRGB(190, 230, 200)
	else
		holdBtn.Text = "hold: OFF — one-shot unlock"
		holdBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 44)
		holdBtn.TextColor3 = Color3.fromRGB(170, 174, 182)
	end
end

		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

	end)
