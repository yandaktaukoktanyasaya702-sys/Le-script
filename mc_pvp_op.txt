if _G.MC_PVP_ATTR_GEN then
	_G.MC_PVP_ATTR_GEN = _G.MC_PVP_ATTR_GEN + 1
else
	_G.MC_PVP_ATTR_GEN = 1
end

local old = PG:FindFirstChild("MC_PVP_ATTR")
if old then
	old:Destroy()
end

local function readBool(name, default)
	local value = LP:GetAttribute(name)
	if type(value) == "boolean" then
		return value
	end
	return default
end

local function readNumber(name, default)
	local value = LP:GetAttribute(name)
	if type(value) == "number" then
		return value
	end
	return default
end

local CFG = {
	killaura = true,
	aimbot = true,
	esp = true,
	reach = math.clamp(readNumber("cheatReach", 6), 3, 6),
}
_G.MC_PVP_CFG = CFG

local function setAttr(name, value)
	pcall(function()
		LP:SetAttribute(name, value)
	end)
end

local function apply()
	setAttr("cheatKillaura", CFG.killaura)
	setAttr("cheatAimbot", CFG.aimbot)
	setAttr("cheatEsp", CFG.esp)
	setAttr("cheatReach", CFG.reach)
end

local function active()
	return _G.MC_PVP_ATTR_GEN == GEN
end

local buttons = {}

local function refresh()
	for _, item in ipairs(buttons) do
		local on = item.get()
		item.button.BackgroundColor3 = on and Color3.fromRGB(32, 92, 72) or Color3.fromRGB(30, 30, 36)
		item.button.TextColor3 = on and Color3.fromRGB(180, 255, 220) or Color3.fromRGB(170, 170, 180)
		item.button.Text = string.format("%s  %s", on and "[ON]" or "[--]", item.text)
	end
	status.Text = string.format("KA %s | AIM %s | ESP %s | reach %.1f", CFG.killaura and "ON" or "OFF", CFG.aimbot and "ON" or "OFF", CFG.esp and "ON" or "OFF", CFG.reach)
end

local function addToggle(text, key, get, set)
	local button = Instance.new("TextButton")
	button.Name = text:gsub("%s+", "")
	button.Size = UDim2.new(1, 0, 0, 27)
	button.BackgroundColor3 = Color3.fromRGB(30, 30, 36)
	button.BorderSizePixel = 0
	button.Font = Enum.Font.Gotham
	button.TextSize = 12
	button.TextXAlignment = Enum.TextXAlignment.Left
	button.Text = text
	button.Parent = body

	local pad = Instance.new("UIPadding")
	pad.PaddingLeft = UDim.new(0, 8)
	pad.Parent = button

	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, 5)
	c.Parent = button

	button.MouseButton1Click:Connect(function()
		set(not get())
		apply()
		refresh()
	end)

	buttons[#buttons + 1] = { button = button, text = text, key = key, get = get, set = set }
end

	CFG.reach = math.clamp(value, 3, 6)
	reachValue.Text = string.format("%.1f", CFG.reach)
	apply()
	refresh()
end

local hidden = false
hideBtn.MouseButton1Click:Connect(function()
	hidden = not hidden
	body.Visible = not hidden
	frame.Size = UDim2.fromOffset(286, hidden and 36 or 226)
	hideBtn.Text = hidden and "+" or "-"
end)

UIS.InputBegan:Connect(function(input, processed)
	if not active() or processed or UIS:GetFocusedTextBox() then
		return
	end
	if input.KeyCode == Enum.KeyCode.RightShift then
		frame.Visible = not frame.Visible
		return
	end
	for _, item in ipairs(buttons) do
		if item.key and input.KeyCode == item.key then
			item.set(not item.get())
			apply()
			refresh()
			return
		end
	end
end)

LP.CharacterAdded:Connect(function()
	task.wait(0.5)
	if active() then
		apply()
	end
end)

apply()
refresh()
